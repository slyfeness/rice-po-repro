  actual_teacher_state_dict_sha256=identity['state_dict_sha256'],peak_allocated=torch.cuda.max_memory_allocated(),peak_reserved=torch.cuda.max_memory_reserved()))
 if not all(c['within_bf16_tolerance'] for c in checks):raise ValueError('vLLM sampled versus HF conditional probabilities failed; no BF16 fallback')
else:raise ValueError('unknown stage')
